Configuring DHCPv4
==================
BR(config)#ip dhcp excluded-address 172.16.1.1 172.16.1.10

BR(config)#ip dhcp pool UserPool
BR(dhcp-config)#network 172.16.1.0 255.255.255.0
BR(dhcp-config)#default-router 172.16.1.1
BR(dhcp-config)#dns-server 8.8.8.8
BR(dhcp-config)#exit

Configuring Stateless DHCPv6
============================
BR(config)#ipv6 dhcp pool Stateless-DHCPv6
BR(config-dhcpv6)#dns-server 2001:4860:4860::8888
BR(config-dhcpv6)#exit

BR(config)#int g0/0/0
BR(config-if)#ipv6 dhcp server Stateless-DHCPv6
BR(config-if)#ipv6 nd other-config-flag
BR(config-if)#exit

BR#show ipv6 dhcp binding

Configuring Stateful DHCPv6
===========================
BR(config)#interface gigabitEthernet 0/0/0

BR(config-if)#ipv6 address 2001:211:A:1::1/64
BR(config-if)#no shutdown 

BR(config)#ipv6 unicast-routing
BR(config)#ipv6 dhcp pool Stateful-DHCPv6
BR(config-dhcpv6)#address prefix 2001:211:A:1::/64
BR(config-dhcpv6)#dns-server 2001:4860:4860::8888
BR(config-dhcpv6)#exit

BR(config)#interface g0/0/0
BR(config-if)#ipv6 dhcp server Stateful-DHCPv6
BR(config-if)#ipv6 nd managed-config-flag 
BR(config-if)#exit
